async function uploadBill() { let file = document.getElementById("billImage").files[0]; if (!file) { alert("Please choose a bill."); return; } document.getElementById("status").innerHTML = "Reading bill..."; let formData = new FormData(); formData.append("file", file); formData.append("apikey", "K81412699488957"); formData.append("language", "eng"); formData.append("isOverlayRequired", "false"); formData.append("OCREngine", "2"); try { const response = await fetch("https://api.ocr.space/parse/image", { method: "POST", headers: { "Accept": "application/json" }, body: formData }) const result = await response.json(); console.log(result); document.getElementById("status").innerHTML = JSON.stringify(result).substring(0,150); alert(JSON.stringify(result, null, 2)); if (result.IsErroredOnProcessing) { document.getElementById("status").innerHTML = "❌ " + result.ErrorMessage; return; } if (result.ParsedResults && result.ParsedResults.length > 0 && result.ParsedResults[0].ParsedText.trim() !== "") { document.getElementById("status").innerHTML = "✅ Bill read successfully"; alert(result.ParsedResults[0].ParsedText); } else { document.getElementById("status").innerHTML = "❌ No text found in the image."; } } catch (err) { alert(err); document.getElementById("status").innerHTML = "❌ " + err; } }
